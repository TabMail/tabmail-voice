    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(json)

  target_compile_definitions(voice-hotkey PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(voice-hotkey PRIVATE nlohmann_json::nlohmann_json user32)
  if(BUILD_TESTING)

    target_link_libraries(voice-screen-access-tests PRIVATE nlohmann_json::nlohmann_json user32 urlmon oleaut32)

      target_compile_options(voice-screen-access-tests PRIVATE /W4 /WX /utf-8)

    add_executable(voice-redactor-tests tests/redactor.cpp)
    target_include_directories(voice-redactor-tests PRIVATE src)

    if(MSVC)
      target_compile_options(voice-redactor-tests PRIVATE /W4 /WX /utf-8)
    endif()
    add_test(NAME voice-redactor COMMAND voice-redactor-tests "${CMAKE_CURRENT_SOURCE_DIR}/../shared/privacy/redaction-cases.json")
    add_executable(voice-ui-fixture tests/ui-fixture.cpp)
    target_compile_definitions(voice-ui-fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)

    target_compile_definitions(voice-privacy-fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(voice-privacy-fixture PRIVATE nlohmann_json::nlohmann_json user32 ole32 oleaut32 uiautomationcore uuid)

    add_test(NAME voice-privacy-provider COMMAND ${NODE_EXECUTABLE}

    add_test(NAME voice-accessibility COMMAND ${NODE_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/accessibility.mjs $<TARGET_FILE:voice-windows> $<TARGET_FILE:voice-ui-fixture>)
