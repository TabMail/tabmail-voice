  # Exercise the exact production worker with synthetic OS/provider boundaries.
  # Reconfigure on dispatcher edits so this cannot silently test stale source.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/system.cpp")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/system.cpp" ACCESSIBILITY_SYSTEM_SOURCE)
  string(FIND "${ACCESSIBILITY_SYSTEM_SOURCE}" "class AccessibilityWorker {" ACCESSIBILITY_WORKER_START)
  if(ACCESSIBILITY_WORKER_START LESS 0)
    message(FATAL_ERROR "AccessibilityWorker test source not found")
  endif()
  string(SUBSTRING "${ACCESSIBILITY_SYSTEM_SOURCE}" ${ACCESSIBILITY_WORKER_START} -1 ACCESSIBILITY_WORKER_SOURCE)
  string(FIND "${ACCESSIBILITY_WORKER_SOURCE}" "\n};" ACCESSIBILITY_WORKER_END)
  if(ACCESSIBILITY_WORKER_END LESS 0)
    message(FATAL_ERROR "AccessibilityWorker test source is incomplete")
  endif()
  math(EXPR ACCESSIBILITY_WORKER_LENGTH "${ACCESSIBILITY_WORKER_END} + 3")
  string(SUBSTRING "${ACCESSIBILITY_WORKER_SOURCE}" 0 ${ACCESSIBILITY_WORKER_LENGTH} ACCESSIBILITY_WORKER_SOURCE)
  configure_file(tests/accessibility-worker.cpp.in accessibility-worker.cpp @ONLY)
  find_package(Threads REQUIRED)
  add_executable(voice-accessibility-worker-tests "${CMAKE_CURRENT_BINARY_DIR}/accessibility-worker.cpp")
  target_include_directories(voice-accessibility-worker-tests PRIVATE tests)
  target_link_libraries(voice-accessibility-worker-tests PRIVATE nlohmann_json::nlohmann_json Threads::Threads voice-core)
  add_test(NAME voice-accessibility-worker COMMAND voice-accessibility-worker-tests)
  set_tests_properties(voice-accessibility-worker PROPERTIES TIMEOUT 15)
